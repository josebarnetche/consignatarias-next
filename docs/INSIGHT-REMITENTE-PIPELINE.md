```javascript
// After fetching MAG data, INSERT to Supabase instead of just JSON
async function storeRemitentes(consigSlug, magData) {
  const entries = magData.entries.map(e => ({
    consignataria_slug: consigSlug,
    mag_id: magData.magId,
    remitente: e.remitente,
    localidad: e.localidad,
    provincia: e.provincia,
    cabezas: e.cabezas,
    remate_date: new Date(), // or parse from period
  }));
  
  await supabase.from('remitente_entries')
    .upsert(entries, { onConflict: 'consignataria_slug,remitente,localidad,remate_date' });
}
```
